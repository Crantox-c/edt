import { DarkTheme, DefaultTheme, ThemeProvider as NavigationThemeProvider } from "expo-router";
import * as SplashScreen from "expo-splash-screen";

import { AnimatedSplashOverlay } from "@/components/animated-icon";
import AppTabs from "@/components/app-tabs";
import { ThemeProvider, useTheme } from "@/hooks/use-theme";

SplashScreen.preventAutoHideAsync();

function RootNavigator() {
  const theme = useTheme();
  const navTheme = theme.isDark ? DarkTheme : DefaultTheme;

  return (
    <NavigationThemeProvider value={navTheme}>
      <AnimatedSplashOverlay />
      <AppTabs />
    </NavigationThemeProvider>
  );
}

export default function TabLayout() {
  return (
    <ThemeProvider>
      <RootNavigator />
    </ThemeProvider>
  );
}

function formatLastSync(lastSync: number | null): string {
  if (!lastSync) return "--:--";
  return new Intl.DateTimeFormat("fr-FR", {
    timeZone: "Europe/Paris",
    hour: "2-digit",
    minute: "2-digit",
  }).format(new Date(lastSync));
}

import { ThemedText } from "@/components/themed-text";
import { ThemedView } from "@/components/themed-view";
import { GlassSurfaceContainer } from "@/components/ui/glass-surface";
import { Spacing } from "@/constants/theme";
import { useTheme } from "@/hooks/use-theme";
import { formatDayHeader, getParisDateString, getTodayParisDateString } from "@/lib/dates";
import { Course } from "@/lib/ics";
import { CourseCard } from "./course-card";

export type ListViewProps = {
  courses: Course[];
  onSelectCourse: (course: Course) => void;
  modifiedKeys?: Set<string>;
};

export function ListView({ courses, onSelectCourse, modifiedKeys }: ListViewProps) {
  const theme = useTheme();
  const glacier = theme.glacier;
  const todayStr = getTodayParisDateString();

  // Grouper les cours par date pour une liste chronologique
  const sections = useMemo(() => {
    const map = new Map<string, Course[]>();
    courses.forEach((c) => {
      const d = getParisDateString(c.start);
      if (!map.has(d)) map.set(d, []);
      map.get(d)!.push(c);
    });

    return Array.from(map.entries()).map(([dateStr, list]) => ({
      dateStr,
      title: formatDayHeader(dateStr),
      isToday: dateStr === todayStr,
      data: list,
    }));
  }, [courses, todayStr]);

  if (courses.length === 0) {
    return (
      <ThemedView type="backgroundElement" style={styles.emptyContainer}>
        <ThemedText type="default">Aucun cours disponible</ThemedText>
        <ThemedText type="small" themeColor="textSecondary">
          Tirez vers le bas pour synchroniser.
        </ThemedText>
      </ThemedView>
    );
  }

  return (
    <View style={styles.container}>
      {sections.map((section) => (
        <View key={section.dateStr} style={styles.sectionContainer}>
          <View style={styles.sectionHeader}>
            <ThemedText
              type="smallBold"
              style={[
                styles.sectionTitle,
                section.isToday && { color: glacier[500] },
              ]}
            >
              {section.title}
            </ThemedText>
            {section.isToday && (
              <View
                style={[
                  styles.todayBadge,
                  { backgroundColor: glacier[100] },
                ]}
              >
                <ThemedText
                  type="smallBold"
                  style={{ color: glacier[700], fontSize: 10 }}
                >
                  {"Aujourd'hui"}
                </ThemedText>
              </View>
            )}
            <ThemedText
              type="small"
              themeColor="textSecondary"
              style={styles.sectionCount}
            >
              {section.data.length} cours
            </ThemedText>
          </View>

          <GlassSurfaceContainer spacing={Spacing.two} style={styles.coursesGroup}>
            {section.data.map((course) => (
              <CourseCard
                key={course.key}
                course={course}
                isModified={modifiedKeys?.has(course.key)}
                onPress={() => onSelectCourse(course)}
              />
            ))}
          </GlassSurfaceContainer>
        </View>
      ))}
    </View>
  );
}

export function getSubjectTint(subject: string): TintName {
  let h = 2166136261;
  const str = subject.toUpperCase();
  for (let i = 0; i < str.length; i++) {
    h = Math.imul(h ^ str.charCodeAt(i), 16777619);
  }
  h ^= h >>> 13;
  h = Math.imul(h, 0x5bd1e995);
  h ^= h >>> 15;

  return TINTS[index];
}

import { useEffect, useState } from "react";
import { AccessibilityInfo, Platform } from "react-native";
import { isLiquidGlassAvailable } from "expo-glass-effect";

/**
 * Hook pour dÃ©terminer si le Liquid Glass natif d'iOS est disponible et utilisable.
 * RÃ¨gle AGENTS.md :
 * Avant tout GlassView, tester isLiquidGlassAvailable().
 * Sinon (iOS < 26, accessibilitÃ© Â« rÃ©duire la transparence Â», Android, aperÃ§u web) :
 * repli sur une surface opaque ou semi-opaque propre, jamais un verre imitÃ©.
 */
export function useLiquidGlass(): boolean {
  const [reduceTransparency, setReduceTransparency] = useState<boolean>(false);

  useEffect(() => {
    let mounted = true;

    // Sur iOS, vÃ©rifier le rÃ©glage systÃ¨me "RÃ©duire la transparence"
    if (Platform.OS === "ios") {
      AccessibilityInfo.isReduceTransparencyEnabled()
        .then((enabled) => {
          if (mounted) setReduceTransparency(enabled);
        })
        .catch(() => {});

      const sub = AccessibilityInfo.addEventListener(
        "reduceTransparencyChanged",
        (enabled: boolean) => {
          setReduceTransparency(enabled);
        }
      );

      return () => {
        mounted = false;
        sub.remove();
      };
    }

    return () => {
      mounted = false;
    };
  }, []);

  // Sur plateforme non-iOS ou si la transparence est rÃ©duite, repli direct
  if (Platform.OS !== "ios" || reduceTransparency) {
    return false;
  }

  // VÃ©rification de la disponibilitÃ© du composant natif expo-glass-effect
  try {
    return isLiquidGlassAvailable();
  } catch {
    return false;
  }
}

        const saved = await AsyncStorage.getItem(THEME_STORAGE_KEY);
        if (saved && (saved === "nuit" || saved === "clair" || saved === "oled" || saved === "auto")) {
          if (mounted) setThemeModeState(saved as ThemeMode);
        }
      } catch (e) {
        console.warn("Erreur chargement thÃ¨me:", e);
      }
    }
    loadTheme();
    return () => {
      mounted = false;
    };
  }, []);

  const setThemeMode = useCallback((mode: ThemeMode) => {
    setThemeModeState(mode);
    AsyncStorage.setItem(THEME_STORAGE_KEY, mode).catch((e) => {
      console.warn("Erreur sauvegarde thÃ¨me:", e);
    });
  }, []);

  // RÃ©soudre le thÃ¨me effectif
  const theme = useMemo<AppTheme>(() => {
    if (themeMode === "auto") {
      return systemScheme === "dark" ? Themes.nuit : Themes.clair;
    }
    return Themes[themeMode];
  }, [themeMode, systemScheme]);

  return (
    <ThemeContext.Provider value={{ theme, themeMode, setThemeMode }}>
      {children}
    </ThemeContext.Provider>
  );
}

 * Hook principal retournant le thÃ¨me actif (avec ses tokens et Ã©chelles de couleurs).
 */
export function useTheme(): AppTheme {
  const context = useContext(ThemeContext);
  return context.theme;
}

  return useContext(ThemeContext);
